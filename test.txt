CREATE TABLE users (id SERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, email TEXT UNIQUE, age INTEGER CHECK (age >= 18), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
CREATE TABLE products (product_id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, price NUMERIC(10,2) CHECK (price > 0), category VARCHAR(50), in_stock BOOLEAN DEFAULT true);
CREATE TABLE orders (order_id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), product_id INTEGER REFERENCES products(product_id), quantity INTEGER NOT NULL CHECK (quantity > 0), order_date DATE DEFAULT CURRENT_DATE);
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_products_category ON products(category);
